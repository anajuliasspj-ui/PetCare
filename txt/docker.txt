# REDE (só uma vez)
docker network create vetcare-rede

# MYSQL (suba primeiro)
docker build -f banco/Dockerfile.mysql -t vetcare-mysql .
docker run -d --name vetcare-mysql --network vetcare-rede -p 3306:3306 vetcare-mysql

# FLASK
docker build --no-cache -f banco/Dockerfile.flask -t vetcare .
docker run -d --name vetcare --network vetcare-rede -p 5000:5000 vetcare

# EXCLUIR E REFAZER
docker rm -f vetcare
docker rm -f vetcare-mysql

# VERIFICAR O BANCO
docker exec -it vetcare-mysql mysql -u root -p

USE vetcare;
SELECT id, user, email, tipo FROM usuarios;
SELECT responsavel, cpf, data_nascimento_responsavel FROM estoque;
